11 — Ball Velocity

Reading the sketch

class Ball {
    constructor(x, y, vx, vy, d, c) {
        this.x = x;
        this.y = y;
        this.vx = vx;          // velocity: how much x changes per move()
        this.vy = vy;          // velocity: how much y changes per move()
        this.diameter = d;
        this.col = c;
    }

    move() {
        this.x += this.vx;     // step position forward by velocity
        this.y += this.vy;
    }

    display() {
        noStroke();
        fill(this.col);
        circle(this.x, this.y, this.diameter);
    }
}

let balls;
const bg = 230;

function setup() {
    createCanvas(640, 360);

    balls = [
        new Ball(100, 100,  2,  1, 60,  color(255, 0, 100, 200)),   // vx, vy now supplied too
        new Ball(300, 150, -1,  2, 100, color(0, 200, 100, 180)),
        ...
    ];
}

function draw() {
    background(bg);                       // erase last frame, every frame (sketch 02)

    for (let i = 0; i < balls.length; i++) {
        balls[i].move();                  // step this ball forward
        balls[i].display();               // draw it at its new position
    }
}

setup() now only builds the array — it no longer displays anything. The loop that used to run once, in setup() (sketch 10), now runs every frame, in draw(): each ball moves a little, then is drawn, 60 times a second.
