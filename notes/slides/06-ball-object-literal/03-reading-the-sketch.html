06 — A Ball Object

Reading the sketch

let ball;                 // declared globally, no value yet
const bg = 230;

function setup() {
    createCanvas(640, 360);
    background(bg);

    let col = color(255, 0, 100, 150);   // local — only exists during this call to setup()

    ball = {
        x: 320,
        y: 180,
        diameter: 150,
        col: col
    };
}

function draw() {
}

function mousePressed() {
    ball.x = mouseX;
    ball.y = mouseY;

    noStroke();
    fill(ball.col);
    circle(ball.x, ball.y, ball.diameter);
}

function keyPressed() {
    background(bg);
}

The separate diameter variable and fixed position are gone; everything about the ball — position, size, colour — now lives in one ball object, read and written via dot notation.
