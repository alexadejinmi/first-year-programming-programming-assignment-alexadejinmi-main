07 — A createBall() Function

Reading the sketch

function createBall(x, y, d, c) {
    let b = {
        x: x,
        y: y,
        diameter: d,
        col: c
    };
    return b;               // hand the built object back to the caller
}

let ball;
const bg = 230;

function setup() {
    createCanvas(640, 360);
    background(bg);

    let col = color(255, 0, 100, 150);
    ball = createBall(320, 180, 150, col);   // was: ball = { x: 320, y: 180, ... }
}

function draw() {
}

function mousePressed() {
    ball.x = mouseX;
    ball.y = mouseY;

    noStroke();
    fill(ball.col);
    circle(ball.x, ball.y, ball.diameter);
}

function keyPressed() {
    background(bg);
}

The object-literal construction that was written inline in setup() (sketch 06) has moved into createBall(); setup() now calls it with the same four values as arguments, instead of writing the literal directly. Everything else — mousePressed(), keyPressed() — is unchanged.
