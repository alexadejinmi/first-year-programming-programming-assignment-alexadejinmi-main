10 — An Array of Balls

Reading the sketch

let balls;                 // will hold an array of Ball objects, once setup() runs

function setup() {
    createCanvas(640, 360);
    background(bg);

    balls = [                                          // an array literal…
        new Ball(100, 100, 60,  color(255, 0, 100, 200)),  // …of five Ball instances,
        new Ball(300, 150, 100, color(0, 200, 100, 180)),  // each with its own position,
        new Ball(500, 80,  40,  color(0, 100, 255, 160)),  // size and colour
        new Ball(200, 280, 80,  color(255, 200, 0, 200)),
        new Ball(450, 270, 100, color(200, 0, 255, 150))
    ];

    textAlign(CENTER, CENTER);

    for (let i = 0; i < balls.length; i++) {   // once per ball, i = 0, 1, 2, 3, 4
        balls[i].display();                     // draw that ball
        fill(0);
        text("Ball " + i, balls[i].x, balls[i].y);   // label it with its index
    }
}

function draw() {
}

mousePressed()/keyPressed() are gone — this sketch is driven entirely by setup(), which now builds five balls into one array instead of one ball into one variable, and loops over that array once to display and label every ball it holds.
